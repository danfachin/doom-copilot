// Doom Copilot CVars.
//
// dc_autospawn_squad — when 1, deploys the 3-bot squad automatically
//   on every map load. Set from the launcher preset's extra_args:
//     +set dc_autospawn_squad 1
//   Leave at 0 for manual summoning via F5/F6/F7/F8/F9 keybinds.

user bool dc_autospawn_squad = false;

// dc_debug — squad telemetry level.
//   0 = off (default)
//   1 = spawn, state transitions, damage, deaths (event-driven)
//   2 = +bot_tick at 2 Hz per live bot (per-persona state/pos/enemy)
// All events write `[DC]` JSON lines to the GZDoom console; the launcher
// redirects console output to the logfile so they live alongside
// hearth-logger's `[HL]` events. Grep the logfile post-run.
user int dc_debug = 0;

// dc_debug_brawler_kite — when 1, Brawler emits a [DC]kite_phase JSON
// event on each oscillation transition (dash-in <-> dash-out). Lets
// you trace the kite cycle without re-enabling the dc_debug=1 firehose.
// Cheap — at most ~22 events/min per Brawler (2 transitions per 1.6s
// cycle, only emits when phase actually changes).
user bool dc_debug_brawler_kite = false;
